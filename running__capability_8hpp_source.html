36// RobotHandle::run_capability<ActionT>() already knows ActionT at the call
37// site (it's the template parameter), so RobotHandle only needs a
38// *non-template* handle to hold in its single per-capability-type map for
39// the ActionT-independent parts (current state, stop(), timeout
40// enforcement) -- RunningCapabilityBase. run() itself, which needs
41// ActionT::Goal, is not part of that virtual interface: RobotHandle
42// dynamic_casts down to the concrete RunningCapability<ActionT> (verifying
43// this capability_type hasn't previously been run() with a *different*

52template<typename OutcomeT>
53easyfleet::CapabilityState outcome_to_state(OutcomeT outcome)
54{
55 switch (outcome) {
56 case OutcomeT::SUCCEEDED: return easyfleet::CapabilityState::SUCCEEDED;
57 case OutcomeT::ABORTED: return easyfleet::CapabilityState::ABORTED;
58 case OutcomeT::CANCELED: return easyfleet::CapabilityState::CANCELED;
59 case OutcomeT::REJECTED: return easyfleet::CapabilityState::REJECTED;
60 case OutcomeT::TIMEOUT: return easyfleet::CapabilityState::TIMEOUT;
61 case OutcomeT::SERVER_UNAVAILABLE: return easyfleet::CapabilityState::UNREACHABLE;
62 }
63 return easyfleet::CapabilityState::UNREACHABLE;
64}

123 void run(const typename ActionT::Goal & goal, std::chrono::seconds timeout)
124 {
125 // A new run() call means a new goal supersedes whatever was in flight
126 // (server-side preemption -- see RobotHandle::run_capability()'s own
127 // doc comment). The *client*-side outcome of that superseded goal can
128 // still arrive after this point, though: its own request() callback
129 // below is still registered and will fire (typically with an
130 // ABORTED-flavored outcome) once the server settles it. Without this
131 // generation guard, that late callback would stomp state_ with the
132 // *old* goal's outcome after the new goal has already started (or
133 // even after it has already finished) -- exactly the race that
134 // produced spurious FAILED results for a robot whose goal was
135 // preempted mid-mission. Only the request() call that is still the
136 // *current* one when its callback fires is allowed to update state_.
137 uint64_t generation;
138 {
139 std::lock_guard<std::mutex> lock(settle_mutex_);
140 generation = ++generation_;
141 state_.store(easyfleet::CapabilityState::RUNNING);
142 }
143 deadline_ = std::chrono::steady_clock::now() + timeout;
144 cancel_deadline_.reset();
145 status_.set_status(robot_name_, "Running " + capability_type_);
146
147 client_->request(
148 goal,
149 [this, generation](const typename Client::Response & response) {
150 settle(generation, outcome_to_state(response.outcome));
151 });
152 }
153
154 void stop() override
155 {
156 if (state_.load() != easyfleet::CapabilityState::RUNNING) {
157 return;
158 }
159 request_cancel();
160 }
161
162 easyfleet::CapabilityState state() const override
163 {
164 return state_.load();
165 }
166
167 void check_timeout(bool alive) override
168 {
169 if (state_.load() != easyfleet::CapabilityState::RUNNING) {
170 return;
171 }
172 const auto now = std::chrono::steady_clock::now();
173 if (!alive) {
174 // No heartbeat: the result may never come. Give up on this goal.
175 settle(std::nullopt, easyfleet::CapabilityState::UNREACHABLE);
176 } else if (cancel_deadline_ && now >= *cancel_deadline_) {
177 // Canceled, but the capability never reported back.
178 settle(std::nullopt, easyfleet::CapabilityState::TIMEOUT);
179 } else if (!cancel_deadline_ && now >= deadline_) {
180 request_cancel();
181 }
182 }
183
184private:
187 void request_cancel()
188 {
189 if (cancel_deadline_) {
190 return;
191 }
192 cancel_deadline_ = std::chrono::steady_clock::now() + kCancelGrace;
193 client_->cancel();
194 }
195
200 void settle(std::optional<uint64_t> generation, easyfleet::CapabilityState new_state)
201 {
202 {
203 std::lock_guard<std::mutex> lock(settle_mutex_);
204 if (generation) {
205 if (*generation != generation_) {
206 return;
207 }
208 } else {
209 ++generation_;
210 }
211 state_.store(new_state);
212 }
213 status_.set_status(robot_name_, capability_type_ + " -> " + easyfleet::to_string(new_state));
214 std::ostringstream out;
215 out << " " << easyfleet_mission_manager::ansi::dim << "[" << robot_name_ << "/" <<
216 capability_type_ << "] " << easyfleet_mission_manager::ansi::reset <<
217 "finished with outcome " << easyfleet_mission_manager::ansi::magenta <<
218 easyfleet::to_string(new_state) << easyfleet_mission_manager::ansi::reset;
219 easyfleet_mission_manager::safe_print(out.str());
220 }
221
222 std::string robot_name_;
223 std::string capability_type_;
224 easyfleet_mission_manager::StatusMarkerPublisher & status_;
225 std::atomic<easyfleet::CapabilityState> state_{easyfleet::CapabilityState::IDLE};
226 std::chrono::steady_clock::time_point deadline_;
228 std::optional<std::chrono::steady_clock::time_point> cancel_deadline_;
231 std::mutex settle_mutex_;
232 uint64_t generation_{0};
235 typename Client::SharedPtr client_;
236};
