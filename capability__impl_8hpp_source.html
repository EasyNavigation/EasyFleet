41template<typename ActionServerT>
42Capability<ActionServerT>::Capability(
43 const std::string & capability_name,
44 const rclcpp::NodeOptions & options)
45: rclcpp_lifecycle::LifecycleNode(capability_name, options),
46 capability_name_(capability_name)
47{
48 this->declare_parameter(detail::kCapabilitiesFileParam, std::string());
49 action_server_ = std::make_shared<ActionServerT>(*this, capability_name_);
50 // Goals are only accepted while the capability is ACTIVE.
51 action_server_->disable_goals();
52
53 robot_name_ = detail::strip_leading_slash(this->get_namespace());
54 resolved_action_name_ = this->get_node_base_interface()->resolve_topic_or_service_name(
55 action_server_->get_action_name(), /*is_service=*/ false);
56
57 RCLCPP_INFO(
58 this->get_logger(),
59 "Starting capability '%s' on robot '%s' (action: %s).",
60 capability_name_.c_str(),
61 detail::robot_label(this->get_namespace()).c_str(),
62 resolved_action_name_.c_str());
63}

98template<typename ActionServerT>
99typename Capability<ActionServerT>::CallbackReturn
100Capability<ActionServerT>::on_configure(const rclcpp_lifecycle::State &)
101{
102 capabilities_pub_ = this->create_publisher<easyfleet_interfaces::msg::CapabilityDescription>(
103 "/capabilities", rclcpp::QoS(1).reliable().transient_local());
104 status_pub_ = this->create_publisher<easyfleet_interfaces::msg::CapabilityStatus>(
105 "/capabilities_status", rclcpp::QoS(1).reliable());
106 return CallbackReturn::SUCCESS;
107}

109template<typename ActionServerT>
110typename Capability<ActionServerT>::CallbackReturn
111Capability<ActionServerT>::on_activate(const rclcpp_lifecycle::State & previous_state)
112{
113 const auto file_path = this->get_parameter(detail::kCapabilitiesFileParam).as_string();
114 if (file_path.empty()) {
115 RCLCPP_ERROR(
116 this->get_logger(),
117 "Capability '%s': the '%s' parameter must be set to a capabilities JSON file path.",
118 capability_name_.c_str(), detail::kCapabilitiesFileParam);
119 return CallbackReturn::FAILURE;
120 }
121
122 std::ifstream file(file_path);
123 if (!file.is_open()) {
124 RCLCPP_ERROR(
125 this->get_logger(),
126 "Capability '%s': could not open capabilities file '%s'.",
127 capability_name_.c_str(), file_path.c_str());
128 return CallbackReturn::FAILURE;
129 }
130 std::ostringstream contents;
131 contents << file.rdbuf();
132
133 // Activates capabilities_pub_/status_pub_ so publish() below actually
134 // sends, only once the file is known to be valid; a subclass overriding
135 // on_activate must call this or the lifecycle-managed publishers stay muted.
136 rclcpp_lifecycle::LifecycleNode::on_activate(previous_state);
137
138 easyfleet_interfaces::msg::CapabilityDescription description_msg;
139 description_msg.robot = robot_name_;
140 description_msg.capability = capability_name_;
141 description_msg.action_name = resolved_action_name_;
142 description_msg.description_json = contents.str();
143 capabilities_pub_->publish(description_msg);
144
145 heartbeat_timer_ = this->create_wall_timer(
146 std::chrono::seconds(1), std::bind(&Capability::publish_heartbeat, this));
147 action_server_->enable_goals();
148
149 return CallbackReturn::SUCCESS;
150}

152template<typename ActionServerT>
153typename Capability<ActionServerT>::CallbackReturn
154Capability<ActionServerT>::on_deactivate(const rclcpp_lifecycle::State & previous_state)
155{
156 // Rejects new goals and waits for the running one to stop.
157 action_server_->disable_goals();
158 stop_heartbeat();
159 rclcpp_lifecycle::LifecycleNode::on_deactivate(previous_state);
160 return CallbackReturn::SUCCESS;
161}

184template<typename ActionServerT>
185void Capability<ActionServerT>::publish_heartbeat()
186{
187 easyfleet_interfaces::msg::CapabilityStatus msg;
188 msg.robot = robot_name_;
189 msg.capability = capability_name_;
190 msg.action_name = resolved_action_name_;
191 msg.busy = action_server_->is_active();
192 status_pub_->publish(msg);
193}

195template<typename ActionServerT>
196void Capability<ActionServerT>::stop_heartbeat()
197{
198 if (heartbeat_timer_) {
199 heartbeat_timer_->cancel();
200 heartbeat_timer_.reset();
201 }
