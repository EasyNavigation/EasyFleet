42template<typename ActionServerT>
43Capability<ActionServerT>::Capability(
44 const std::string & capability_name,
45 const rclcpp::NodeOptions & options)
46: rclcpp_lifecycle::LifecycleNode(capability_name, options),
47 capability_name_(capability_name)
48{
49 this->declare_parameter(detail::kCapabilitiesFileParam, std::string());
50 const auto prefix = detail::sanitize_parameter_name(capability_name_);
51 order_history_param_ = prefix + ".order_history";
52 link_timeout_param_ = prefix + ".link_timeout";
53 on_link_lost_param_ = prefix + ".on_link_lost";
54 this->declare_parameter(order_history_param_, 20);
55 this->declare_parameter(link_timeout_param_, 0.0);
56 this->declare_parameter(on_link_lost_param_, std::string("continue"));
57 order_history_ = std::make_unique<OrderHistory>(
58 static_cast<std::size_t>(std::max<int64_t>(1,
59 this->get_parameter(order_history_param_).as_int())));
60
61 action_server_ = std::make_shared<ActionServerT>(*this, capability_name_);
62 // Goals are only accepted while the capability is ACTIVE.
63 action_server_->disable_goals();
64
65 robot_name_ = detail::strip_leading_slash(this->get_namespace());
66 resolved_action_name_ = this->get_node_base_interface()->resolve_topic_or_service_name(
67 action_server_->get_action_name(), /*is_service=*/ false);
68
69 RCLCPP_INFO(
70 this->get_logger(),
71 "Starting capability '%s' on robot '%s' (action: %s).",
72 capability_name_.c_str(),
73 detail::robot_label(this->get_namespace()).c_str(),
74 resolved_action_name_.c_str());
75
76 action_server_->set_goal_accepted_callback(
77 [this](const rclcpp_action::GoalUUID & uuid, const std::string & order_id) {
78 const auto id = order_history_->accepted(
79 uuid, order_id, resolved_action_name_, this->now());
80 RCLCPP_INFO(
81 this->get_logger(), "Capability '%s': order '%s' accepted.",
82 capability_name_.c_str(), id.c_str());
83 });
84 // The action server's own status topic: the states of every goal it holds.
85 goal_status_sub_ = this->create_subscription<action_msgs::msg::GoalStatusArray>(
86 resolved_action_name_ + "/_action/status",
87 rclcpp::QoS(rclcpp::KeepLast(1)).reliable().transient_local(),
88 [this](const action_msgs::msg::GoalStatusArray & msg) {
89 order_history_->update(msg, this->now());
90 });
91}

93template<typename ActionServerT>
94Capability<ActionServerT>::~Capability()
95{
96 action_server_->set_goal_accepted_callback(nullptr);
97 stop_heartbeat();
98 // Join the worker while the derived action server still exists.
99 action_server_->stop();
100}

133template<typename ActionServerT>
134easyfleet_interfaces::msg::CapabilityStatus Capability<ActionServerT>::get_status() const
135{
136 easyfleet_interfaces::msg::CapabilityStatus msg;
137 msg.robot = robot_name_;
138 msg.capability = capability_name_;
139 msg.action_name = resolved_action_name_;
140 msg.busy = action_server_->is_active();
141 if (const auto order = order_history_->latest()) {
142 msg.order_id = order->order_id;
143 msg.order_state = order->state;
144 }
145 msg.link_state = link_watchdog_.state();
146 return msg;
147}

149template<typename ActionServerT>
150typename Capability<ActionServerT>::CallbackReturn
151Capability<ActionServerT>::on_configure(const rclcpp_lifecycle::State &)
152{
153 const auto on_link_lost = this->get_parameter(on_link_lost_param_).as_string();
154 if (on_link_lost != "continue" && on_link_lost != "cancel") {
155 RCLCPP_ERROR(
156 this->get_logger(), "Capability '%s': '%s' must be \"continue\" or \"cancel\", not \"%s\".",
157 capability_name_.c_str(), on_link_lost_param_.c_str(), on_link_lost.c_str());
158 return CallbackReturn::FAILURE;
159 }
160 cancel_on_link_lost_ = on_link_lost == "cancel";
161
162 capabilities_pub_ = this->create_publisher<easyfleet_interfaces::msg::CapabilityDescription>(
163 "/capabilities", rclcpp::QoS(1).reliable().transient_local());
164 status_pub_ = this->create_publisher<easyfleet_interfaces::msg::CapabilityStatus>(
165 "/capabilities_status", rclcpp::QoS(1).reliable());
166 get_order_srv_ = this->create_service<easyfleet_interfaces::srv::GetOrder>(
167 "~/get_order",
168 [this](
169 const std::shared_ptr<easyfleet_interfaces::srv::GetOrder::Request> request,
170 std::shared_ptr<easyfleet_interfaces::srv::GetOrder::Response> response) {
171 on_get_order(request, response);
172 });
173 return CallbackReturn::SUCCESS;
174}

176template<typename ActionServerT>
177void Capability<ActionServerT>::on_get_order(
178 const std::shared_ptr<easyfleet_interfaces::srv::GetOrder::Request> request,
179 std::shared_ptr<easyfleet_interfaces::srv::GetOrder::Response> response)
180{
181 if (const auto order = order_history_->find(request->order_id)) {
182 response->found = true;
183 response->order = *order;
184 }

187template<typename ActionServerT>
188typename Capability<ActionServerT>::CallbackReturn
189Capability<ActionServerT>::on_activate(const rclcpp_lifecycle::State & previous_state)
190{
191 const auto file_path = this->get_parameter(detail::kCapabilitiesFileParam).as_string();
192 if (file_path.empty()) {
193 RCLCPP_ERROR(
194 this->get_logger(),
195 "Capability '%s': the '%s' parameter must be set to a capabilities JSON file path.",
196 capability_name_.c_str(), detail::kCapabilitiesFileParam);
197 return CallbackReturn::FAILURE;
198 }
199
200 std::ifstream file(file_path);
201 if (!file.is_open()) {
202 RCLCPP_ERROR(
203 this->get_logger(),
204 "Capability '%s': could not open capabilities file '%s'.",
205 capability_name_.c_str(), file_path.c_str());
206 return CallbackReturn::FAILURE;
207 }
208 std::ostringstream contents;
209 contents << file.rdbuf();
210
211 // Activates capabilities_pub_/status_pub_ so publish() below actually
212 // sends, only once the file is known to be valid; a subclass overriding
213 // on_activate must call this or the lifecycle-managed publishers stay muted.
214 rclcpp_lifecycle::LifecycleNode::on_activate(previous_state);
215
216 easyfleet_interfaces::msg::CapabilityDescription description_msg;
217 description_msg.robot = robot_name_;
218 description_msg.capability = capability_name_;
219 description_msg.action_name = resolved_action_name_;
220 description_msg.description_json = contents.str();
221 capabilities_pub_->publish(description_msg);
222
223 heartbeat_timer_ = this->create_wall_timer(
224 std::chrono::seconds(1), std::bind(&Capability::publish_heartbeat, this));
225
226 const double link_timeout = this->get_parameter(link_timeout_param_).as_double();
227 link_watchdog_.reset(link_timeout);
228 if (link_watchdog_.monitored()) {
229 fleet_heartbeat_sub_ = this->create_subscription<easyfleet_interfaces::msg::FleetHeartbeat>(
230 "/fleet_heartbeat", rclcpp::QoS(1).best_effort(),
231 [this](const easyfleet_interfaces::msg::FleetHeartbeat &) {
232 link_watchdog_.heard(LinkWatchdog::Clock::now());
233 });
234 // A few checks per timeout, at most one a second.
235 const auto period = std::chrono::duration<double>(std::clamp(link_timeout / 4.0, 0.01, 1.0));
236 link_timer_ = this->create_wall_timer(
237 std::chrono::duration_cast<std::chrono::nanoseconds>(period),
238 std::bind(&Capability::check_link, this));
239 RCLCPP_INFO(
240 this->get_logger(),
241 "Capability '%s': watching /fleet_heartbeat (timeout %.2f s, on loss: %s).",
242 capability_name_.c_str(), link_timeout, cancel_on_link_lost_ ? "cancel" : "continue");
243 }
244 action_server_->enable_goals();
245
246 return CallbackReturn::SUCCESS;
247}

250typename Capability<ActionServerT>::CallbackReturn
251Capability<ActionServerT>::on_deactivate(const rclcpp_lifecycle::State & previous_state)
252{
253 // Rejects new goals and waits for the running one to stop.
254 action_server_->disable_goals();
255 stop_heartbeat();
256 rclcpp_lifecycle::LifecycleNode::on_deactivate(previous_state);
257 return CallbackReturn::SUCCESS;

289template<typename ActionServerT>
290void Capability<ActionServerT>::check_link()
291{
292 const auto before = link_watchdog_.state();
293 const auto now = link_watchdog_.update(LinkWatchdog::Clock::now());
294 if (now == before) {
295 return;
296 }
297 using easyfleet_interfaces::msg::CapabilityStatus;
298 if (now == CapabilityStatus::LINK_LOST) {
299 const bool busy = action_server_->is_active();
300 RCLCPP_WARN(
301 this->get_logger(), "Capability '%s': link to the fleet lost (no /fleet_heartbeat)%s.",
302 capability_name_.c_str(),
303 cancel_on_link_lost_ && busy ? ", stopping the running goal" : "");
304 if (cancel_on_link_lost_) {
305 action_server_->halt_active_goal();
306 }
307 } else if (now == CapabilityStatus::LINK_OK && before == CapabilityStatus::LINK_LOST) {
308 RCLCPP_INFO(
309 this->get_logger(), "Capability '%s': link to the fleet recovered.",
310 capability_name_.c_str());
311 }
312}

314template<typename ActionServerT>
315void Capability<ActionServerT>::stop_heartbeat()
316{
317 if (heartbeat_timer_) {
318 heartbeat_timer_->cancel();
319 heartbeat_timer_.reset();
320 }
321 if (link_timer_) {
322 link_timer_->cancel();
323 link_timer_.reset();
324 }
325 fleet_heartbeat_sub_.reset();
326}
